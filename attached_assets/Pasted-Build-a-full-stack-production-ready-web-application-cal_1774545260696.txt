Theme: Dark luxury — deep charcoal backgrounds with electric indigo/violet accents. Think Notion meets Linear meets a high-end fintech product.

Color Palette (use CSS variables throughout):
  --bg-primary: #0A0A0F
  --bg-secondary: #111118
  --bg-card: #16161F
  --bg-elevated: #1C1C28
  --accent-primary: #6C63FF (electric indigo)
  --accent-secondary: #A78BFA (soft violet)
  --accent-glow: rgba(108, 99, 255, 0.15)
  --text-primary: #F0F0FF
  --text-secondary: #9898B8
  --text-muted: #5A5A7A
  --border-subtle: rgba(108, 99, 255, 0.12)
  --border-active: rgba(108, 99, 255, 0.4)
  --success: #34D399
  --warning: #FBBF24
  --danger: #F87171
  --gradient-hero: linear-gradient(135deg, #6C63FF 0%, #A78BFA 50%, #C084FC 100%)

Typography:
  - Display/Headings: "Syne" (Google Fonts) — geometric, distinctive
  - Body: "DM Sans" (Google Fonts) — clean, readable
  - Mono/Code/Dates: "JetBrains Mono" (Google Fonts)

Animations & Motion (use Framer Motion):
  - Page transitions: fade + slide up (0.3s ease-out)
  - Cards: staggered entrance with 0.05s delay per item
  - Hover on cards: translateY(-2px) + subtle box-shadow glow
  - Buttons: scale(0.97) on press + ripple effect
  - Sidebar: smooth collapse/expand with width transition
  - Loading states: skeleton shimmer with gradient sweep
  - Modal open: scale from 0.95 to 1 + fade
  - Number counters: animate count-up on dashboard load
  - Background: very subtle animated gradient mesh on auth pages
  - Toast notifications: slide in from top-right

Global UI Components:
  - Cards with border: 1px solid var(--border-subtle), background var(--bg-card), border-radius 12px
  - On hover: border color changes to var(--border-active), subtle indigo glow box-shadow
  - Buttons: primary (gradient bg), ghost (transparent + border), danger (red tint)
  - Inputs: dark bg, subtle border, glowing focus ring in accent color
  - Badges/Tags: pill shape, colored backgrounds with matching text
  - Avatars: gradient fallback with initials when no photo
  - Empty states: illustrated SVG placeholders with helpful copy
  - Scrollbars: custom styled, thin, accent-colored

🗄️ DATABASE SCHEMA (Prisma)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

model User {
  id            String   @id @default(cuid())
  name          String?
  email         String   @unique
  emailVerified DateTime?
  image         String?
  password      String?
  plan          String   @default("free") // free, pro, team
  timezone      String   @default("UTC")
  createdAt     DateTime @default(now())
  contacts      Contact[]
  tasks         Task[]
  reminders     Reminder[]
  aiChats       AiChat[]
}

model Contact {
  id            String   @id @default(cuid())
  userId        String
  user          User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  name          String
  email         String?
  phone         String?
  location      String?
  company       String?
  role          String?
  linkedinUrl   String?
  twitterUrl    String?
  website       String?
  avatarUrl     String?
  whereMet      String?
  introducedBy  String?
  topicsDiscussed String?
  notes         String?
  createdAt     DateTime @default(now())
  updatedAt     DateTime @updatedAt
  tags          ContactTag[]
  interactions  Interaction[]
  tasks         Task[]
  reminders     Reminder[]
  files         File[]
  aiSummaries   AiSummary[]
}

model ContactTag {
  id        String  @id @default(cuid())
  contactId String
  contact   Contact @relation(fields: [contactId], references: [id], onDelete: Cascade)
  tag       String
  color     String  @default("#6C63FF")
}

model Interaction {
  id          String   @id @default(cuid())
  contactId   String
  contact     Contact  @relation(fields: [contactId], references: [id], onDelete: Cascade)
  userId      String
  type        String   // call, meeting, email, note, dm, event
  summary     String
  source      String   @default("manual") // manual, calendar, gmail
  occurredAt  DateTime @default(now())
  createdAt   DateTime @default(now())
  files       File[]
}

model Task {
  id          String   @id @default(cuid())
  contactId   String?
  contact     Contact? @relation(fields: [contactId], references: [id], onDelete: SetNull)
  userId      String
  user        User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  title       String
  description String?
  dueDate     DateTime?
  priority    String   @default("medium") // low, medium, high
  status      String   @default("pending") // pending, done
  createdAt   DateTime @default(now())
  completedAt DateTime?
  reminder    Reminder?
}

model Reminder {
  id          String   @id @default(cuid())
  userId      String
  user        User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  contactId   String?
  contact     Contact? @relation(fields: [contactId], references: [id], onDelete: SetNull)
  taskId      String?  @unique
  task        Task?    @relation(fields: [taskId], references: [id])
  message     String
  remindAt    DateTime
  sent        Boolean  @default(false)
  createdAt   DateTime @default(now())
}

model File {
  id            String       @id @default(cuid())
  contactId     String?
  contact       Contact?     @relation(fields: [contactId], references: [id], onDelete: SetNull)
  interactionId String?
  interaction   Interaction? @relation(fields: [interactionId], references: [id])
  userId        String
  fileUrl       String
  fileName      String
  fileType      String
  createdAt     DateTime     @default(now())
}

model AiSummary {
  id          String   @id @default(cuid())
  contactId   String
  contact     Contact  @relation(fields: [contactId], references: [id], onDelete: Cascade)
  userId      String
  summaryText String   @db.Text
  createdAt   DateTime @default(now())
}

model AiChat {
  id        String   @id @default(cuid())
  userId    String
  user      User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  messages  Json     // array of {role, content, timestamp}
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
}

🔌 API ROUTES (Next.js API Routes)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Auth:
  POST /api/auth/signup         → Create user account
  POST /api/auth/[...nextauth]  → NextAuth handler (Google + credentials)

Contacts:
  GET    /api/contacts              → List all (with search, filter, sort query params)
  POST   /api/contacts              → Create new contact
  GET    /api/contacts/[id]         → Get single contact with all related data
  PUT    /api/contacts/[id]         → Update contact
  DELETE /api/contacts/[id]         → Delete contact
  POST   /api/contacts/[id]/tags    → Add tag
  DELETE /api/contacts/[id]/tags/[tagId] → Remove tag

Interactions:
  GET    /api/interactions          → Global timeline (all contacts)
  POST   /api/interactions          → Add new interaction
  PUT    /api/interactions/[id]     → Update interaction
  DELETE /api/interactions/[id]     → Delete interaction

Tasks:
  GET    /api/tasks                 → List all tasks (filter by status, due date)
  POST   /api/tasks                 → Create task
  PUT    /api/tasks/[id]            → Update task (including marking complete)
  DELETE /api/tasks/[id]            → Delete task

Reminders:
  GET    /api/reminders             → List upcoming reminders
  POST   /api/reminders             → Create reminder
  PUT    /api/reminders/[id]        → Update reminder
  DELETE /api/reminders/[id]        → Delete reminder

AI:
  POST   /api/ai/brief              → Generate contact brief (accepts contactId)
  POST   /api/ai/draft-email        → Draft follow-up email (accepts contactId, context)
  POST   /api/ai/chat               → General AI assistant chat (accepts messages array + optional contactId)
